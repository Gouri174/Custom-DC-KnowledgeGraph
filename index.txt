<script>
const PROXY = 'http://localhost:5050';
const ns = 'http://www.w3.org/2000/svg';

const RING_COLORS  = ['#185FA5', '#1D9E75', '#BA7517', '#7F77DD'];
const RING_FILLS   = ['#185FA5', '#E1F5EE', '#FAEEDA', '#EEEDFE'];
const RING_STROKES = ['#0C447C', '#0F6E56', '#854F0B', '#534AB7'];
const RING_GAP     = 200; // px between rings

async function loadGraph() {
  const dcid = document.getElementById('dcid-input').value.trim();
  const depth = parseInt(document.getElementById('depth-slider').value);
  if (!dcid) return;
  setStatus('Fetching graph (depth ' + depth + ') from India Data Commons...');
  clearGraph();

  try {
    const graphData = await fetch(
      `${PROXY}/graph/${encodeURIComponent(dcid)}?depth=${depth}`
    ).then(r => r.json());

    if (graphData.error) throw new Error(graphData.error);

    // Flatten into nodes by depth level
    const nodeMap = {};   // id -> { id, depth, out, parent }
    const edges = [];

    function walk(g, d, parentId) {
      if (!g || !g.dcid) return;
      if (!nodeMap[g.dcid] || nodeMap[g.dcid].depth > d) {
        nodeMap[g.dcid] = { id: g.dcid, depth: d, out: g.out || {}, parent: parentId };
      }

      const out = g.out || {};
      Object.keys(out).forEach(prop => {
        if (prop === 'dcid') return;
        (out[prop] || []).forEach(v => {
          const target = v.dcid || v.name;
          if (!target) return;
          edges.push({ source: g.dcid, target, label: prop });
          if (!nodeMap[target]) {
            nodeMap[target] = { id: target, depth: d + 1, out: {}, parent: g.dcid };
          }
        });
      });

      (g.children || []).forEach(child => {
        if (child.dcid) {
          edges.push({ source: g.dcid, target: child.dcid, label: 'expands_to' });
          walk(child, d + 1, g.dcid);
        }
      });
    }

    walk(graphData, 0, null);

    const nodes = Object.values(nodeMap);
    const uniqueEdges = dedupEdges(edges);

    // Group nodes by depth
    const byDepth = {};
    nodes.forEach(n => {
      if (!byDepth[n.depth]) byDepth[n.depth] = [];
      byDepth[n.depth].push(n);
    });

    const maxDepth = Math.max(...Object.keys(byDepth).map(Number));
    const totalNodes = nodes.length;
    const totalEdges = uniqueEdges.length;

    setStatus(`Loaded — ${totalNodes} nodes, ${totalEdges} edges · depth=${depth} · Source: indiadatacommons.org`);

    // Canvas: radius of outermost ring + padding
    const outerR = (maxDepth + 1) * RING_GAP;
    const cx = outerR + 120, cy = outerR + 120;
    const W = cx * 2, H = cy * 2;

    // Assign positions: radial tree layout
    // Root at center, each depth on its own ring
    // Within each ring, nodes are evenly spaced angularly
    // BUT: group children near their parent's angle for clarity

    // First pass: assign angles based on parent
    assignAngles(nodeMap, byDepth, maxDepth);

    // Compute (x, y) from angle and depth
    nodes.forEach(n => {
      if (n.depth === 0) { n.x = cx; n.y = cy; }
      else {
        const r = n.depth * RING_GAP;
        n.x = cx + r * Math.cos(n.angle);
        n.y = cy + r * Math.sin(n.angle);
      }
    });

    renderGraph(nodes, uniqueEdges, nodeMap, graphData.dcid, W, H, cx, cy, maxDepth);

  } catch(e) {
    setStatus('Error: ' + e.message + ' — Is the proxy running? Run: python app.py');
  }
}

function assignAngles(nodeMap, byDepth, maxDepth) {
  // Root gets angle 0
  const root = Object.values(nodeMap).find(n => n.depth === 0);
  if (root) root.angle = 0;

  // For each depth level, group nodes by parent and assign angles
  for (let d = 1; d <= maxDepth; d++) {
    const nodesAtDepth = byDepth[d] || [];
    if (!nodesAtDepth.length) continue;

    // Group by parent
    const byParent = {};
    nodesAtDepth.forEach(n => {
      const p = n.parent || '__root__';
      if (!byParent[p]) byParent[p] = [];
      byParent[p].push(n);
    });

    // For each parent, spread children around parent's angle
    const parents = Object.keys(byParent);
    const totalParents = parents.length;

    // Divide full circle among parents
    parents.forEach((parentId, pi) => {
      const children = byParent[parentId];
      const parentNode = nodeMap[parentId];
      const parentAngle = parentNode ? (parentNode.angle || 0) : (pi / totalParents) * 2 * Math.PI;

      // Each parent gets a sector of the circle
      // Sector width: divide 2π by number of distinct parents at this level
      const sectorWidth = (2 * Math.PI) / Math.max(totalParents, 1);
      const startAngle = parentAngle - sectorWidth / 2;

      children.forEach((child, ci) => {
        if (children.length === 1) {
          child.angle = parentAngle;
        } else {
          child.angle = startAngle + (ci / (children.length - 1)) * sectorWidth;
        }
      });
    });
  }
}

function dedupEdges(edges) {
  const seen = new Set();
  return edges.filter(e => {
    const key = `${e.source}|||${e.target}|||${e.label}`;
    if (seen.has(key)) return false;
    seen.add(key); return true;
  });
}

function renderGraph(nodes, edges, nodeMap, rootId, W, H, cx, cy, maxDepth) {
  const svg = document.getElementById('graph-svg');
  svg.innerHTML = '';
  svg.setAttribute('viewBox', `0 0 ${W} ${H}`);
  svg.setAttribute('width', W);
  svg.setAttribute('height', H);

  // Draw concentric ring guides (subtle)
  for (let d = 1; d <= maxDepth; d++) {
    const r = d * RING_GAP;
    const circle = document.createElementNS(ns, 'circle');
    circle.setAttribute('cx', cx); circle.setAttribute('cy', cy);
    circle.setAttribute('r', r);
    circle.setAttribute('fill', 'none');
    circle.setAttribute('stroke', '#f0f0f0');
    circle.setAttribute('stroke-width', 1);
    circle.setAttribute('stroke-dasharray', '4,4');
    svg.appendChild(circle);

    // Ring label
    const lt = document.createElementNS(ns, 'text');
    lt.setAttribute('x', cx + r - 8); lt.setAttribute('y', cy - 6);
    lt.setAttribute('font-size', 9); lt.setAttribute('fill', '#ccc');
    lt.setAttribute('font-family', 'Segoe UI, sans-serif');
    lt.setAttribute('text-anchor', 'end');
    lt.textContent = `depth ${d}`;
    svg.appendChild(lt);
  }

  // Defs: arrowhead
  const defs = document.createElementNS(ns, 'defs');
  const m = document.createElementNS(ns, 'marker');
  m.setAttribute('id','arr'); m.setAttribute('viewBox','0 0 10 10');
  m.setAttribute('refX',9); m.setAttribute('refY',5);
  m.setAttribute('markerWidth',5); m.setAttribute('markerHeight',5);
  m.setAttribute('orient','auto-start-reverse');
  const mp = document.createElementNS(ns,'path');
  mp.setAttribute('d','M2 1L8 5L2 9'); mp.setAttribute('fill','none');
  mp.setAttribute('stroke','#ccc'); mp.setAttribute('stroke-width',1.5);
  mp.setAttribute('stroke-linecap','round');
  m.appendChild(mp); defs.appendChild(m); svg.appendChild(defs);

  // Draw edges
  edges.forEach(e => {
    const s = nodeMap[e.source], t = nodeMap[e.target];
    if (!s || !t) return;

    const sx = s.depth === 0 ? cx : s.x;
    const sy = s.depth === 0 ? cy : s.y;
    const tx = t.depth === 0 ? cx : t.x;
    const ty = t.depth === 0 ? cy : t.y;

    // Curved edge using quadratic bezier for readability
    const midX = (sx + tx) / 2 + (ty - sy) * 0.08;
    const midY = (sy + ty) / 2 + (sx - tx) * 0.08;

    const path = document.createElementNS(ns, 'path');
    path.setAttribute('d', `M${sx},${sy} Q${midX},${midY} ${tx},${ty}`);
    path.setAttribute('fill', 'none');
    path.setAttribute('stroke', '#ddd');
    path.setAttribute('stroke-width', 1);
    path.setAttribute('marker-end', 'url(#arr)');
    svg.appendChild(path);

    // Edge label at midpoint
    if (e.label && e.label !== 'expands_to') {
      const lt = document.createElementNS(ns,'text');
      lt.setAttribute('x', midX); lt.setAttribute('y', midY - 3);
      lt.setAttribute('text-anchor','middle');
      lt.setAttribute('font-size', 8);
      lt.setAttribute('fill', '#bbb');
      lt.setAttribute('font-family', 'Segoe UI, sans-serif');
      const lbl = e.label.length > 16 ? e.label.slice(0,16)+'…' : e.label;
      lt.textContent = lbl;
      svg.appendChild(lt);
    }
  });

  // Draw nodes
  nodes.forEach(n => {
    const d = Math.min(n.depth, RING_COLORS.length - 1);
    const isRoot = n.id === rootId;
    const nx = n.depth === 0 ? cx : n.x;
    const ny = n.depth === 0 ? cy : n.y;
    const r = isRoot ? 18 : 10;

    const g = document.createElementNS(ns,'g');
    g.style.cursor = 'pointer';

    const circle = document.createElementNS(ns,'circle');
    circle.setAttribute('cx', nx); circle.setAttribute('cy', ny);
    circle.setAttribute('r', r);
    circle.setAttribute('fill', RING_FILLS[d]);
    circle.setAttribute('stroke', RING_STROKES[d]);
    circle.setAttribute('stroke-width', isRoot ? 2.5 : 1.5);
    g.appendChild(circle);

    // Label below node
    const shortId = n.id.length > 20 ? n.id.slice(0,20)+'…' : n.id;
    const lt = document.createElementNS(ns,'text');
    lt.setAttribute('x', nx); lt.setAttribute('y', ny + r + 12);
    lt.setAttribute('text-anchor','middle');
    lt.setAttribute('font-size', isRoot ? 11 : 9);
    lt.setAttribute('font-weight', isRoot ? '600' : '400');
    lt.setAttribute('fill', isRoot ? '#185FA5' : '#555');
    lt.setAttribute('font-family','Segoe UI, sans-serif');
    lt.textContent = shortId;
    g.appendChild(lt);

    // Hover
    const container = document.getElementById('graph-container');
    g.addEventListener('mouseenter', evt => {
      const tt = document.getElementById('tooltip');
      const out = n.out || {};
      const props = Object.keys(out).filter(k=>k!=='dcid').slice(0,5)
        .map(k => `${k}: ${(out[k]||[]).map(v=>v.name||v.dcid||'').filter(Boolean).slice(0,2).join(', ')}`)
        .join('\n');
      tt.textContent = n.id + (props ? '\n\n' + props : '');
      tt.style.display = 'block';
      const rect = container.getBoundingClientRect();
      const evRect = evt.target.getBoundingClientRect();
      tt.style.left = (evRect.left - rect.left + 16) + 'px';
      tt.style.top  = (evRect.top  - rect.top  + 16) + 'px';
      circle.setAttribute('stroke-width', isRoot ? 4 : 2.5);
      circle.setAttribute('r', r + 2);
    });
    g.addEventListener('mouseleave', () => {
      document.getElementById('tooltip').style.display = 'none';
      circle.setAttribute('stroke-width', isRoot ? 2.5 : 1.5);
      circle.setAttribute('r', r);
    });

    g.addEventListener('click', () => {
      const out = n.out || {};
      const rows = [
        {k:'DCID', v: n.id},
        {k:'Depth', v: n.depth},
        {k:'Parent', v: n.parent || 'root'}
      ];
      Object.keys(out).filter(k=>k!=='dcid').forEach(prop => {
        const vals = (out[prop]||[]).map(v=>v.name||v.dcid||'').filter(Boolean);
        if (vals.length) rows.push({k: prop, v: vals.join(', ')});
      });
      showDetail(n.id, rows);
    });

    svg.appendChild(g);
  });
}

function showDetail(title, rows) {
  const p = document.getElementById('detail-panel');
  let h = `<h3>${esc(title)}</h3>`;
  rows.forEach(r => { h += `<div class="prop-row"><span class="prop-key">${esc(r.k)}</span><span class="prop-val">${esc(r.v)}</span></div>`; });
  p.innerHTML = h; p.classList.add('show');
  p.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
}

function clearGraph() {
  const svg = document.getElementById('graph-svg');
  svg.innerHTML = '';
  svg.setAttribute('viewBox','0 0 900 900');
  svg.removeAttribute('width'); svg.removeAttribute('height');
  document.getElementById('detail-panel').classList.remove('show');
  document.getElementById('tooltip').style.display = 'none';
}

function esc(s) { return String(s||'').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;'); }
function setStatus(t) { document.getElementById('status').textContent = t; }

document.getElementById('dcid-input').addEventListener('keydown', e => { if (e.key==='Enter') loadGraph(); });
</script>
